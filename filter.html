async function loadFilter() { const filterMount = document.getElementById( "filterMount" ); try { const response = await fetch( "/Benefits/filter.html" ); if (!response.ok) { throw new Error( "filter.html tapılmadı" ); } filterMount.innerHTML = await response.text(); const companyFilter = document.getElementById( "companyFilter" ); const positionFilter = document.getElementById( "positionFilter" ); const continueButton = document.getElementById( "continueButton" ); const filterError = document.getElementById( "filterError" ); const savedCompany = localStorage.getItem( "leg-selected-company" ); const savedPosition = localStorage.getItem( "leg-selected-position" ); if (savedCompany) { companyFilter.value = savedCompany; } if (savedPosition) { positionFilter.value = savedPosition; } function updateButton() { const ready = companyFilter.value && positionFilter.value; continueButton.disabled = !ready; filterError.style.display = "none"; } companyFilter.addEventListener( "change", updateButton ); positionFilter.addEventListener( "change", updateButton ); updateButton(); continueButton.addEventListener( "click", function() { if ( !companyFilter.value || !positionFilter.value ) { filterError.style.display = "block"; return; } const companyName = companyFilter.options[ companyFilter.selectedIndex ].text; const positionName = positionFilter.options[ positionFilter.selectedIndex ].text; localStorage.setItem( "leg-selected-company", companyFilter.value ); localStorage.setItem( "leg-selected-company-name", companyName ); localStorage.setItem( "leg-selected-position", positionFilter.value ); localStorage.setItem( "leg-selected-position-name", positionName ); window.location.href = "/Benefits/home.html"; } ); } catch (error) { console.error(error); filterMount.innerHTML = `
⚠️ filter.html yüklənmədi.
`; } } loadFilter();
